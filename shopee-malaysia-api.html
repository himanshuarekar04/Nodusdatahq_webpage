Python Request Example

import requests

api_url = "https://api.nodusdatahq.com/v1/shopee/product"
headers = {"Authorization": "Bearer YOUR_NODUSDATA_API_KEY"}
params = {
    "market": "my",
    "domain": "shopee.com.my",
    "item_id": "123456789",
    "shop_id": "98765432"
}

response = requests.get(api_url, headers=headers, params=params)
data = response.json()
print("Product Title:", data.get("name"))
print("Price (MYR (RM)):", data.get("price"))
